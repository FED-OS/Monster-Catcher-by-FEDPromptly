function tileInFrontOfPlayer() {
  const deltas = { up: [0, -1], down: [0, 1], left: [-1, 0], right: [1, 0] };
  const [dc, dr] = deltas[player.facing];
  return { col: player.col + dc, row: player.row + dr };
}
